% Voltage Source Network (Kirchhoff's Current Law)
% Demonstrates: CircuiTikz with multiple voltage sources, resistors, and currents.
% Compile with: pdflatex main.tex

\documentclass[margin=3mm]{standalone}
\usepackage[european]{circuitikz}
\begin{document}
\begin{circuitikz}
    \draw
       % --- Outer loop with R1, R3, and voltage sources E1/E2 ---
       (0,0) to [R, l_=$R_1$] (2,0)
       to [short] (6,0)
       to [R, l_=$R_3$] (6,4)
       to [short] (6,5)
       to [short] (0,5)
       to [V=$E_1$, i=$i_1$] (0,2.5)
       to [V<=$E_2$] (0,0.5)
       to [short] (0,0)
       % --- Inner branch with E3 and R2 ---
       (4,0) to [V, l_=$E_3$] (4,2.5)
       to [R, l_=$R_2$] (4,4.5)
       to [short] (4,5);
\end{circuitikz}
\end{document}
