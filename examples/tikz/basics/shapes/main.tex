% Basic Shapes in TikZ
% Demonstrates: rectangles, parabolas, circles, and ellipses with styling.
% Compile with: pdflatex main.tex

\documentclass[12pt]{article}
\usepackage{tikz}
\pagestyle{empty}


\begin{document}
\begin{tikzpicture}

% --- Rectangle (method 1: explicit corners) ---
\draw (0,0) -- (10,0) -- (10,10) -- (0,10) -- (0, 0);

% --- Rectangle (method 2: shorthand) ---
\draw (0,0) rectangle (10,10);

% --- Parabola with dashed style ---
\draw[dashed] (0,0) parabola (10,10);

% --- Circle with color ---
\draw[green] (5,5) circle (7.1);

% --- Ellipse with color ---
\draw[blue] (5,5) ellipse (1 and 7.1);

\end{tikzpicture}
\end{document}
