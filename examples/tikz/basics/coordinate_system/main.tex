% Coordinate System Example
% Demonstrates: axes, grid, tick marks, and labeled coordinates in TikZ.
% Compile with: pdflatex main.tex

\documentclass{article}
\usepackage{tikz}
\usetikzlibrary{arrows.meta}

\begin{document}
\begin{tikzpicture}[thick, scale=1, every node/.style={scale=1.2}]

% --- Grid ---
\draw[lightgray, thin] (-4,-4) grid (4,4);

% --- Axes ---
\draw[-{Stealth}, very thick] (-4.5, 0) -- (4.5, 0) node[right] {$x$};
\draw[-{Stealth}, very thick] (0, -4.5) -- (0, 4.5) node[above] {$y$};

% --- Tick marks on x-axis ---
\foreach \x in {-4,-3,...,-1,1,2,...,4} {
    \draw (\x, 0.1) -- (\x, -0.1) node[below, font=\small] {$\x$};
}

% --- Tick marks on y-axis ---
\foreach \y in {-4,-3,...,-1,1,2,...,4} {
    \draw (0.1, \y) -- (-0.1, \y) node[left, font=\small] {$\y$};
}

% --- Origin label ---
\node[below left, font=\small] at (0,0) {$0$};

% --- Plot some points ---
\filldraw[blue] (2,3) circle (2pt) node[above right] {$(2, 3)$};
\filldraw[red]  (-3,1) circle (2pt) node[above left] {$(-3, 1)$};
\filldraw[green!60!black] (1,-2) circle (2pt) node[below right] {$(1, -2)$};

\end{tikzpicture}
\end{document}
