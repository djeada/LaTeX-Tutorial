% Simple Tree
% Demonstrates: basic tree structure with circle nodes and child branches.
% Compile with: pdflatex main.tex

\documentclass[12pt]{article}
\usepackage{tikz}
\pagestyle{empty}
\begin{document}
\begin{tikzpicture} [every node/.style={circle, draw=black}]

% --- Root node with children ---
% TikZ tree syntax: child{ node{...} } creates branches automatically
\node{1}
	child{ node{2}
		child{node{4}}
		child{node{5}}
		child{node{6}}
	}
	child{node{3}}
;
\end{tikzpicture}
\end{document}
