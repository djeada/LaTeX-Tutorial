% Venn Diagram: Intersection (A ∩ B)
% Demonstrates: scope/clip technique to fill only the overlapping region.
% Compile with: pdflatex main.tex

\documentclass{article}
\usepackage{tikz}
\usetikzlibrary{calc}

\begin{document}
\begin{tikzpicture}[thick,scale=1, every node/.style={scale=2}]

% --- Bounding box ---
\filldraw[fill=white] (-5,-5) rectangle (8,5);

% --- Circle outlines ---
\draw (0,0) circle (3);
\draw (3,0) circle (3);

% --- Intersection: clip to circle A, then fill circle B ---
\scope 
\clip (0,0) circle (3);
\fill[red] (3,0) circle (3);
\endscope

% --- Label ---
\node[draw] at (-3,4) {$A \cap B$};

\end{tikzpicture}
\end{document}
