% Venn Diagram: Union (A ∪ B)
% Demonstrates: overlapping circles with fill to show set union.
% Compile with: pdflatex main.tex

\documentclass{article}
\usepackage{tikz}
\usetikzlibrary{calc}

\begin{document}
\begin{tikzpicture}[thick,scale=1, every node/.style={scale=2}]

% --- Bounding box ---
\filldraw[fill=white] (-5,-5) rectangle (8,5);

% --- Draw and fill both circles (union = both colored) ---
\fill[red] (0,0) circle (3);
\fill[red] (3,0) circle (3);

% --- Circle outlines ---
\draw (0,0) circle (3);
\draw (3,0) circle (3);

% --- Label ---
\node[draw] at (-3,4) {$A \cup B$};

\end{tikzpicture}
\end{document}
